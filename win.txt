# ============================================================
#  FS Print - Workshop Claude Code - Bootstrap para Windows
#
#  Uso (PowerShell COMO ADMINISTRADOR):
#      irm https://datricas.github.io/kit/win | iex
#
#  Para ver que haria sin tocar nada:
#      $env:FSPRINT_CHECK=1; irm https://datricas.github.io/kit/win | iex
#
#  Instala WSL2 (Linux adentro de Windows) y despues corre adentro
#  el mismo instalador que usan las Mac.
#
#  NOTA TECNICA: todo va adentro de una funcion y no hay ni un 'exit'.
#  Cuando un script llega por 'iex' corre en la sesion actual, asi que
#  un 'exit' cerraria la ventana de PowerShell del usuario.
# ============================================================

function Invoke-FSPrintBootstrap {
  [CmdletBinding()]
  param([switch]$Check)

  if ($env:FSPRINT_CHECK -eq '1') { $Check = $true }

  # Se guarda y se restaura, para no dejarle la sesion cambiada.
  $eapPrevio = $ErrorActionPreference
  $ErrorActionPreference = 'Stop'

  function Azul($t)  { Write-Host $t -ForegroundColor Cyan }
  function OK($t)    { Write-Host "  [OK]  $t" -ForegroundColor Green }
  function Info($t)  { Write-Host "  ->    $t" -ForegroundColor Gray }
  function Aviso($t) { Write-Host "  [!]   $t" -ForegroundColor Yellow }
  function Error2($t){ Write-Host "  [X]   $t" -ForegroundColor Red }
  function Haria($t) {
    if ($Check) { Write-Host "  [~]   haria: $t" -ForegroundColor Yellow; return $true }
    return $false
  }

  try {
    Write-Host ""
    Azul "==========================================="
    if ($Check) { Azul "  Modo chequeo - no se modifica nada" }
    else        { Azul "  FS Print - Preparando tu Windows" }
    Azul "==========================================="
    Write-Host ""

    # ---- 1. Permisos de administrador ----
    $esAdmin = ([Security.Principal.WindowsPrincipal] `
      [Security.Principal.WindowsIdentity]::GetCurrent()
      ).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)

    if (-not $esAdmin) {
      Error2 "Abri PowerShell como administrador (boton derecho sobre el icono > Ejecutar como administrador) y volve a pegar el comando."
      return
    }
    OK "Permisos de administrador"

    # ---- 2. Version de Windows ----
    $build = [int](Get-CimInstance Win32_OperatingSystem).BuildNumber
    if ($build -lt 19041) {
      Error2 "Tu Windows es muy viejo para WSL2 (build $build, hace falta 19041 o mayor). Actualiza Windows y volve a intentar."
      return
    }
    OK "Windows build $build"

    # ---- 3. WSL2 con Ubuntu ----
    $wslListo = $false
    try {
      $distros = (wsl.exe --list --quiet 2>$null) -replace "`0", ""
      if ($distros -match 'Ubuntu') { $wslListo = $true }
    } catch { $wslListo = $false }

    if (-not $wslListo) {
      if (Haria "instalar WSL2 con Ubuntu (pide reiniciar)") {
        Aviso "Sin WSL no se puede seguir. El resto del chequeo asume que ya estaria instalado."
      } else {
        Info "Instalando WSL2 con Ubuntu. Tarda unos minutos..."
        wsl.exe --install -d Ubuntu
        Write-Host ""
        Azul "  >>> HAY QUE REINICIAR LA COMPUTADORA <<<"
        Write-Host ""
        Write-Host "  Cuando vuelvas a prender:"
        Write-Host "    1. Se abre sola una ventana de Ubuntu: elegi un usuario y una clave."
        Write-Host "       (la clave no se ve mientras la escribis, es normal)"
        Write-Host "    2. Volve a abrir PowerShell como administrador"
        Write-Host "    3. Pega de nuevo el mismo comando"
        Write-Host ""
        return
      }
    } else {
      OK "WSL2 con Ubuntu"
      $arranca = $false
      try {
        $prueba = wsl.exe -d Ubuntu -- echo listo 2>$null
        if ($prueba -match 'listo') { $arranca = $true }
      } catch { $arranca = $false }

      if ($arranca) { OK "Ubuntu responde" }
      else {
        Error2 "Ubuntu esta instalada pero no arranca. Abri la app 'Ubuntu' desde el menu Inicio, crea el usuario y la clave, y volve a intentar."
        return
      }
    }

    # ---- 4. Windows Terminal ----
    if (-not (Get-Command wt.exe -ErrorAction SilentlyContinue)) {
      if (-not (Haria "instalar Windows Terminal")) {
        Info "Instalando Windows Terminal..."
        try {
          winget install --id Microsoft.WindowsTerminal -e --silent `
            --accept-package-agreements --accept-source-agreements | Out-Null
          OK "Windows Terminal"
        } catch {
          Aviso "No se pudo instalar Windows Terminal. Se puede seguir sin el."
        }
      }
    } else {
      OK "Windows Terminal"
    }

    # ---- 5. Fuente con iconos ----
    if (-not (Haria "instalar la fuente Nerd Font")) {
      try {
        winget install --id DEVCOM.JetBrainsMonoNerdFont -e --silent `
          --accept-package-agreements --accept-source-agreements | Out-Null
        OK "Fuente Nerd Font"
      } catch {
        Aviso "La fuente hay que instalarla a mano despues. No frena nada: solo se ven cuadraditos en vez de iconos."
      }
    }

    # ---- 6. El instalador de Linux, adentro de WSL ----
    Write-Host ""
    if (Haria "correr el instalador de Linux adentro de Ubuntu") {
      Write-Host ""
      Azul "==========================================="
      Azul "  Chequeo terminado. No se toco nada."
      Azul "==========================================="
      Write-Host ""
      return
    }

    Azul "  Entrando a Linux para instalar el resto..."
    Write-Host ""
    wsl.exe -d Ubuntu -- bash -lc "curl -fsSL https://datricas.github.io/kit/linux | bash"
    if ($LASTEXITCODE -ne 0) {
      Error2 "El instalador de Linux fallo. Mira el error de arriba y avisa."
      return
    }

    Write-Host ""
    Azul "==========================================="
    OK "Listo."
    Write-Host ""
    Write-Host "  Abri Windows Terminal y escribi:  doctor"
    Write-Host ""
    Azul "==========================================="
    Write-Host ""
  }
  catch {
    Write-Host ""
    Error2 "Error inesperado: $($_.Exception.Message)"
    Write-Host "  Sacale una foto a esto y avisa." -ForegroundColor Gray
    Write-Host ""
  }
  finally {
    $ErrorActionPreference = $eapPrevio
  }
}

Invoke-FSPrintBootstrap
