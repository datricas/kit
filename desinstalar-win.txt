# ============================================================
#  FS Print - Workshop Claude Code - DESINSTALADOR (Windows)
#
#  Uso (PowerShell COMO ADMINISTRADOR):
#      irm https://datricas.github.io/kit/desinstalar-win | iex
#
#  Saca Ubuntu, la tarea de reanudacion y la configuracion de
#  Windows Terminal que puso el kit.
#
#  OJO: borrar Ubuntu borra TODO lo que haya adentro. Si guardaste
#  algo ahi, sacalo antes.
# ============================================================

function Invoke-FSPrintDesinstalar {
  [CmdletBinding()]
  param()

  $eapPrevio = $ErrorActionPreference
  $ErrorActionPreference = 'Stop'

  function Azul($t)  { Write-Host $t -ForegroundColor Cyan }
  function OK($t)    { Write-Host "  [OK]  $t" -ForegroundColor Green }
  function Info($t)  { Write-Host "  ->    $t" -ForegroundColor Gray }
  function Salteo($t){ Write-Host "  .     $t" -ForegroundColor DarkGray }
  function Error2($t){ Write-Host "  [X]   $t" -ForegroundColor Red }

  try {
    Write-Host ""
    Azul "==========================================="
    Azul "  FS Print - Desinstalar de Windows"
    Azul "==========================================="
    Write-Host ""

    $esAdmin = ([Security.Principal.WindowsPrincipal] `
      [Security.Principal.WindowsIdentity]::GetCurrent()
      ).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
    if (-not $esAdmin) {
      Error2 "Abri PowerShell como administrador y volve a pegar el comando."
      return
    }

    Write-Host "  Se va a hacer:" -ForegroundColor White
    Write-Host "    . Borrar Ubuntu, PERO solo si la instalo el kit"
    Write-Host "      (si ya la tenias, o si usas Docker, no se toca nada)"
    Write-Host "    . Sacar la tarea de reanudacion del instalador"
    Write-Host "    . Restaurar la configuracion de Windows Terminal"
    Write-Host ""
    Write-Host "  NO se toca: WSL en si, Windows Terminal, ni la fuente." -ForegroundColor Gray
    Write-Host "  (son funciones de Windows que podes seguir usando)" -ForegroundColor Gray
    Write-Host ""
    $rta = Read-Host "  Seguimos? [s/N]"
    if ($rta -notmatch '^[sSyY]') { Write-Host "  Cancelado."; return }

    # ---- 1. La tarea de reanudacion ----
    Write-Host ""
    Azul "1/3 - Tarea de reanudacion"
    $clave = 'HKCU:\Software\Microsoft\Windows\CurrentVersion\RunOnce'
    if (Get-ItemProperty -Path $clave -Name 'FSPrintKit' -ErrorAction SilentlyContinue) {
      Remove-ItemProperty -Path $clave -Name 'FSPrintKit' -Force
      OK "quitada del arranque"
    } else { Salteo "no estaba" }

    $script = Join-Path $env:LOCALAPPDATA 'fsprint-continuar.ps1'
    if (Test-Path $script) { Remove-Item $script -Force; OK "script de reanudacion" }
    else { Salteo "script de reanudacion - no estaba" }

    # ---- 2. Windows Terminal ----
    Write-Host ""
    Azul "2/3 - Windows Terminal"
    $wt = Get-ChildItem "$env:LOCALAPPDATA\Packages" -Filter 'Microsoft.WindowsTerminal*' `
            -Directory -ErrorAction SilentlyContinue | Select-Object -First 1
    if ($wt) {
      $cfg    = Join-Path $wt.FullName 'LocalState\settings.json'
      $backup = "$cfg.antes-del-kit"
      if (Test-Path $backup) {
        Copy-Item $backup $cfg -Force
        Remove-Item $backup -Force
        OK "configuracion restaurada a como estaba"
      } else {
        Salteo "no hay copia de seguridad - se deja como esta"
        Info "si queres volver al tema original, borra settings.json y"
        Info "Windows Terminal lo recrea al abrir"
      }
    } else { Salteo "no esta instalado" }

    # ---- 3. Ubuntu ----
    Write-Host ""
    Azul "3/3 - Ubuntu"

    $distros = @()
    try {
      $distros = ((wsl.exe --list --quiet 2>$null) -replace "`0", "") -split "`r?`n" |
                 Where-Object { $_.Trim() -ne '' }
    } catch { }

    if ($distros.Count -eq 0) {
      Salteo "no hay ninguna distribucion de Linux instalada"
    } else {
      Write-Host "  Distribuciones que tenes:" -ForegroundColor Gray
      foreach ($d in $distros) { Write-Host "    . $d" -ForegroundColor Gray }
      Write-Host ""

      # Otras distros (Docker, etc.) NUNCA se tocan.
      $otras = $distros | Where-Object { $_.Trim() -ne 'Ubuntu' }
      if ($otras.Count -gt 0) {
        Info "Las que no son Ubuntu no se tocan (Docker y demas quedan como estan)."
        Write-Host ""
      }

      $hayUbuntu = $distros | Where-Object { $_.Trim() -eq 'Ubuntu' }
      if (-not $hayUbuntu) {
        Salteo "no hay ninguna Ubuntu"
      } else {
        # Solo la borramos si la marca dice que la instalamos nosotros.
        $laPusimosNosotros = $false
        try {
          $m = Get-ItemProperty -Path 'HKCU:\Software\Datricas' `
                 -Name 'UbuntuInstaladaPorElKit' -ErrorAction SilentlyContinue
          if ($m -and $m.UbuntuInstaladaPorElKit -eq 1) { $laPusimosNosotros = $true }
        } catch { }

        # Sin marca no sabemos con certeza. En vez de adivinar, juntamos
        # evidencia, se la mostramos y que decida la persona: la marca es
        # nueva, asi que quien instalo antes NUNCA la va a tener.
        if (-not $laPusimosNosotros) {
          Write-Host "  No hay marca del kit. Puede ser porque la instalaste antes" -ForegroundColor Gray
          Write-Host "  de que el instalador empezara a dejarla, o porque Ubuntu ya" -ForegroundColor Gray
          Write-Host "  era tuya. Miremos las pistas:" -ForegroundColor Gray
          Write-Host ""

          # Pista 1: cuando se creo la distribucion
          $creada = $null
          try {
            $lxss = 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Lxss'
            Get-ChildItem $lxss -ErrorAction SilentlyContinue | ForEach-Object {
              $pr = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
              if ($pr.DistributionName -eq 'Ubuntu' -and $pr.BasePath) {
                $ruta = $pr.BasePath -replace '^\\\\\?\\', ''
                if (Test-Path $ruta) { $creada = (Get-Item $ruta).CreationTime }
              }
            }
          } catch { }
          if ($creada) {
            $dias = [math]::Round(((Get-Date) - $creada).TotalDays, 1)
            Write-Host ("    . Ubuntu se creo el {0}  (hace {1} dias)" -f $creada.ToString('dd/MM/yyyy HH:mm'), $dias)
          } else {
            Write-Host "    . No pude averiguar cuando se creo"
          }

          # Pista 2: rastros del instalador del kit
          $rastro = (Test-Path (Join-Path $env:LOCALAPPDATA 'fsprint-continuar.ps1')) -or
                    ($null -ne (Get-ItemProperty -Path 'HKCU:\Software\Microsoft\Windows\CurrentVersion\RunOnce' `
                                  -Name 'FSPrintKit' -ErrorAction SilentlyContinue))
          if ($rastro) {
            Write-Host "    . El instalador del kit SI dejo rastros de haber instalado WSL"
          } else {
            Write-Host "    . No hay rastros del instalador del kit (puede que ya se hayan limpiado)"
          }

          # Pista 3: que mas hay adentro, aparte del kit
          $otrasCosas = ''
          try {
            $otrasCosas = (wsl.exe -d Ubuntu -- bash -lc "ls -A ~ 2>/dev/null | grep -v -E '^\.|fsprint|practica' | head -8 | tr '\n' ' '" 2>$null)
          } catch { }
          if ([string]::IsNullOrWhiteSpace($otrasCosas)) {
            Write-Host "    . Adentro no hay nada tuyo, solo lo del workshop"
          } else {
            Write-Host ("    . Adentro hay cosas tuyas: {0}" -f $otrasCosas.Trim()) -ForegroundColor Yellow
          }

          Write-Host ""
          Write-Host "  Si Ubuntu la instalamos en el workshop, se puede borrar." -ForegroundColor White
          Write-Host "  Si ya la tenias antes, o la usas para Docker: NO." -ForegroundColor White
          Write-Host ""
          $r1 = Read-Host "  La instalo el kit en el workshop? [s/N]"
          if ($r1 -match '^[sSyY]') { $laPusimosNosotros = $true }
          else {
            Salteo "Ubuntu se deja como esta"
            Info "El kit ya se borro de adentro. Ubuntu queda intacta y usable."
          }
        }

        if ($laPusimosNosotros) {
          Write-Host ""
          Write-Host ""
          Write-Host "  >>> Esto borra Ubuntu y TODO lo que tenga adentro <<<" -ForegroundColor Yellow
          Write-Host "      Si guardaste algo ahi despues del workshop, sacalo primero." -ForegroundColor Yellow
          Write-Host ""
          $r2 = Read-Host "  Escribi BORRAR para confirmar (o enter para dejarla)"
          if ($r2 -eq 'BORRAR') {
            wsl.exe --terminate Ubuntu 2>$null | Out-Null
            wsl.exe --unregister Ubuntu
            if ($LASTEXITCODE -eq 0) {
              OK "Ubuntu borrada"
              try { Remove-ItemProperty -Path 'HKCU:\Software\Datricas' `
                      -Name 'UbuntuInstaladaPorElKit' -Force -ErrorAction SilentlyContinue } catch { }
            } else {
              Error2 "No se pudo borrar. Proba a mano: wsl --unregister Ubuntu"
            }
          } else {
            Salteo "Ubuntu se deja como esta"
          }
        }
      }
    }

    Write-Host ""
    Azul "==========================================="
    OK "Listo."
    Write-Host ""
    Write-Host "  Si NO usas WSL para nada mas (ojo con Docker), se puede apagar:" -ForegroundColor Gray
    Write-Host "      dism /online /disable-feature /featurename:Microsoft-Windows-Subsystem-Linux" -ForegroundColor DarkGray
    Write-Host "      dism /online /disable-feature /featurename:VirtualMachinePlatform" -ForegroundColor DarkGray
    Write-Host "  (pide reiniciar. Solo si no lo usas para nada mas.)" -ForegroundColor Gray
    Write-Host ""
    Azul "==========================================="
    Write-Host ""
  }
  catch {
    Write-Host ""
    Error2 "Error: $($_.Exception.Message)"
    Write-Host ""
  }
  finally {
    $ErrorActionPreference = $eapPrevio
  }
}

Invoke-FSPrintDesinstalar
